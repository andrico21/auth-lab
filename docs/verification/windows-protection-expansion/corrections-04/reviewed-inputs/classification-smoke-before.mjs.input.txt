function eq(actual,expected,label){report.checks++;try{assert.deepEqual(copy(actual),copy(expected));}catch{report.failures.push({label,actual:copy(actual),expected:copy(expected)});}}
const ok=(condition,label)=>eq(Boolean(condition),true,label);
let app,independentModel,scope;
function groups(){
  const snapshot=app.protocolSnapshot(),results=app.labModel.pamStatePaths.results.split('.').reduce((value,key)=>value?.[key],snapshot.state),expected={public:{},internal:{},pending:{}},actual={};
  for(const [key,status] of Object.entries(results||{}))expected[status==='not evaluated'?'pending':key==='prelim'||key==='update'?'internal':'public'][key]=status;
  for(const [group,key,summary] of definitions){
    const disclosure=app.querySelector('#pam-configuration-panel [data-pam-disclosure="'+key+'"]');
    if(results===undefined){eq(Boolean(disclosure),false,scope+' no result disclosure before any result map: '+group);actual[group]={};continue;}
    ok(disclosure,scope+' explicit '+group+' disclosure exists');
    if(!disclosure){actual[group]=undefined;continue;}
    eq(disclosure.querySelector('summary')?.textContent,summary,scope+' literal '+group+' summary');
    try{actual[group]=JSON.parse(disclosure.querySelector('pre')?.textContent);}catch(error){report.failures.push({label:scope+' valid '+group+' result JSON',error:String(error)});actual[group]=undefined;}
    eq(actual[group],expected[group],scope+' actual '+group+' classification');
    if(group==='internal'){
      const explanation=disclosure.querySelector('p')?.textContent||'';
      ok(/prelim/.test(explanation)&&/update/.test(explanation)&&/pam_chauthtok/.test(explanation)&&/passes.*inside|inside.*passes/i.test(explanation)&&/not separate application calls/i.test(explanation),scope+' internal passes explained inside enclosing public call');
    }
  }
  if(results!==undefined){eq(Object.hasOwn(actual.public||{},'prelim'),false,scope+' no completed public preliminary API');eq(Object.hasOwn(actual.public||{},'update'),false,scope+' no completed public update API');eq(Object.values(actual.public||{}).includes('not evaluated'),false,scope+' no completed not-evaluated placeholders');}
  const index=snapshot.cursor.eventIndex;
  eq(snapshot.state,reconstructProtocolState(independentModel,index),scope+' real rendered canonical state');
  const observation={scope,index,cursor:copy(snapshot.cursor),groups:copy(actual),canonicalResults:copy(results),step:app.currentStep?.sourceStepId||app.currentStep?.id};report.observations.push(observation);
  return observation;
}

  app.selectStep(index,false);const observed=groups(),sourceIndex=app.protocolSnapshot().cursor.eventIndex,steps=app.steps;
  if(app.labScenarioId==='lab-llog04'){
    const call=steps.findIndex(step=>step.from==='pamChangeApp'&&step.to==='pamChange'&&step.pamEvaluation?.api==='pam_chauthtok'),returned=steps.findIndex(step=>step.from==='pamChange'&&step.to==='pamChangeApp'&&step.pamEvaluation?.api==='pam_chauthtok'&&step.pamEvaluation.aggregateReturn!==undefined);
    if(call>=0&&sourceIndex>=call&&sourceIndex<returned){
      const value=[...(app.querySelector('#pam-configuration-panel .pam-evaluation')?.children||[])].find(row=>row.querySelector('dt')?.textContent==='API aggregate return')?.querySelector('dd')?.textContent;

    }
  }
  return observed;
}

function fullAndRewind(){
  const observed=[];for(let index=0;index<app.steps.length;index++)observed.push(select(index));
  for(let index=app.steps.length-1;index>=0;index--){const after=select(index,'rewind');eq(after.groups,observed[index].groups,scope+' rewind restores classification');eq(after.canonicalResults,observed[index].canonicalResults,scope+' rewind restores canonical results');}
  return observed;
}

  for(const kind of ['field','exact']){
    app.clearAttribute();if(kind==='field')app.selectAttribute(artifact.attributeId,false);else app.heldView.trace(artifactId);
    const observed=[];for(let index=0;index<app.visibleSteps.length;index++)observed.push(select(index,kind));
    for(let index=app.visibleSteps.length-1;index>=0;index--){const after=select(index,kind+'-rewind');eq(after.groups,observed[index].groups,scope+' '+kind+' rewind restores same canonical-prefix classification');}
    views[kind]=observed.map(row=>({index:row.index,groups:row.groups}));
  }
  app.clearAttribute();return views;
}
try{
  let Component;
  if(mode==='emitted'){
    const scripts=[...html.toString().matchAll(/<script>([\s\S]*?)<\/script>/g)];assert.equal(scripts.length,1);
    new vm.Script(scripts[0][1],{filename:'actual-auth-flow-studio.emitted.js'}).runInContext(vm.createContext(sandbox),{timeout:5000});Component=harness.registry.get('auth-flow-studio');
  }else ({AuthFlowStudio:Component}=await import(pathToFileURL(path.join(root,'src/app.js')).href));

  for(const [variant,status] of [['default','PAM_SUCCESS'],['new-policy-denied','PAM_AUTHTOK_ERR'],['update-failed','PAM_AUTHTOK_ERR'],['directory-unreachable','PAM_AUTHINFO_UNAVAIL'],['directory-online','PAM_SUCCESS'],['post-change-account-denied','PAM_SUCCESS']]){
    choose(variant);const observed=fullAndRewind();app.selectStep(app.steps.length-1,false);eq(groups().groups.public?.password,status,scope+' final public password outcome');

  for(const [member,internal,status] of [['change-success',{prelim:'PAM_SUCCESS',update:'PAM_SUCCESS'},'PAM_SUCCESS'],['change-prelim-failure',{prelim:'PAM_AUTHTOK_ERR'},'PAM_AUTHTOK_ERR'],['change-update-failure',{prelim:'PAM_SUCCESS',update:'PAM_AUTHTOK_ERR'},'PAM_AUTHTOK_ERR'],['change-unsupported',{},undefined]]){
    app.selectLabScenario('lab-pam01',{variant:'password-change-required'});app.selectLessonSubfixture(member);independentModel=app.labModel;scope='PAM01/'+member;
    const observed=fullAndRewind();app.selectStep(app.steps.length-1,false);const final=groups();eq(final.groups.internal,internal,scope+' final internal evidence');eq(final.groups.public?.password,status,scope+' final public result only if evaluated');
