const unix=id=>[...LINUXLOGONLAB_SCENARIOS,...LINUXELEVLAB_SCENARIOS].find(model=>model.id===id);
const sourceIndex=(model,key)=>model.steps().findIndex(step=>step.id===model.id+':'+key||step.id===model.id+'-'+key);
const at=(model,index)=>reconstructProtocolState(model,index);
const end=model=>at(model,model.steps().length-1);
const path=(value,key)=>key.split('.').reduce((current,part)=>current?.[part],value);
const findReturn=(model,after,phase,flags)=>model.steps().findIndex((step,index)=>index>after&&step.pamEvaluation?.apiPhase===phase&&step.pamEvaluation.aggregateReturn!==undefined&&(flags===undefined||step.pamEvaluation.flags===flags));
function expectReturn(row,phase,flags,status){expect(row.apiPhase).toBe(phase);expect(row.flags).toBe(flags);expect(row.aggregateReturn).toBe(status);expect(row.moduleHook).toBeNull();expect(row.rawReturn).toBeNull();expect(row.mappedAction).toBeNull();expect(row.accumulatorBefore).toBeNull();expect(row.accumulatorAfter).toBeNull();}

test('WP0 expectation bytes remain frozen and native cached-close golden is not repinned',()=>{
 for(const [name,sha] of Object.entries(frozenHashes)){const bytes=readFileSync(new URL('../docs/pam-foundations/inputs/wp0-original/'+name,import.meta.url));expect(createHash('sha256').update(bytes).digest('hex')).toBe(sha);}
 const native=readFileSync(new URL('../docs/pam-foundations/native-expectations.json',import.meta.url));expect(createHash('sha256').update(native).digest('hex')).toBe('1065337878481993ba2280529a201a7adb166fcaa51aa0287bd80ce5b235d0f1');
 expect(JSON.parse(native).vectors.find(vector=>vector.service==='afs-close-session-jump').calls.find(call=>call.api==='pam_close_session').expected).toBe('PAM_SESSION_ERR');
});
test('password callback error and pam_unix/module aggregate error have distinct source semantics',()=>{
 const model=createPamFoundationsCase('conversation-failure'),callback=sourceIndex(model,'conversation-response'),module=model.steps().findIndex(step=>step.pamEvaluation?.executed&&step.pamEvaluation.apiPhase==='authenticate');
 expect(at(model,callback).workstation.conversationStatus).toBe('PAM_CONV_ERR');expect(at(model,callback).workstation.pamResults.authenticate).toBe('not evaluated');
 expect(model.steps()[module].pamEvaluation.rawReturn).toBe('PAM_AUTHTOK_ERR');expect(model.steps()[module].pamEvaluation.mappedAction).toBe('bad');
 expect(at(model,module).workstation.pamEvaluation.accumulatorAfter).toEqual({impression:'negative',status:'PAM_AUTHTOK_ERR'});
 const result=sourceIndex(model,'return-authenticate-default');expectReturn(model.steps()[result].pamEvaluation,'authenticate',0,'PAM_AUTHTOK_ERR');
 expect(end(model).workstation.pamResults.authenticate).toBe('PAM_AUTHTOK_ERR');expect(end(model).workstation.authenticated).toBe(false);expect(end(model).workstation.session).toBe(false);expect(end(model).workstation.unrelatedSession).toBe('other-user-session-retained');
});
for(const variant of ['default','session-failure'])test('LLOG01 establish/delete results and flags remain separate: '+variant,()=>{
 const model=unix('lab-llog01').withVariant(variant),trigger=sourceIndex(model,variant==='default'?'delete-credentials':'session-deny'),result=findReturn(model,trigger,'setcred','PAM_DELETE_CRED');
 expect(trigger).toBeGreaterThanOrEqual(0);expect(result).toBeGreaterThan(trigger);
 expect(at(model,trigger).workstation.pamResults.setcred).toBe('PAM_SUCCESS');expect(at(model,trigger).workstation.pamResults['setcred-delete']).toBeUndefined();
 const rows=model.steps().slice(trigger+1,result).filter(step=>step.pamEvaluation?.executed&&step.pamEvaluation.apiPhase==='setcred');expect(rows.length).toBeGreaterThan(0);
 for(const step of rows){expect(step.pamEvaluation.flags).toBe('PAM_DELETE_CRED');expect(step.pamEvaluation.moduleHook).toBe('pam_sm_setcred');expect(step.pamEvaluation.type).toBe('auth');expect(step.pamEvaluation.rawReturn).toBe('PAM_SUCCESS');expect(step.pamEvaluation.configurationHash).toBe(model.pamConfiguration.files[0].sha256);}
 expectReturn(model.steps()[result].pamEvaluation,'setcred','PAM_DELETE_CRED','PAM_SUCCESS');
 const state=end(model).workstation;expect(state.pamResults.setcred).toBe('PAM_SUCCESS');expect(state.pamResults['setcred-delete']).toBe('PAM_SUCCESS');expect(state.authenticated).toBe(true);expect(state.admitted).toBe(true);expect(state.session).toBe(false);
 if(variant==='session-failure'){expect(state.pamResults.open).toBe('PAM_SESSION_ERR');expect(state.failure).toBe('PAM_SESSION_ERR');}
});

for(const [id,key,owner] of [['lab-lelev01','command-exit','lelevPAM'],['lab-lelev02','su-return','suPAM']])test('named teardown closes only its configured PAM session: '+id,()=>{
 const model=unix(id),trigger=sourceIndex(model,key),result=findReturn(model,trigger,'close',0),rows=model.steps().slice(trigger+1,result).filter(step=>step.pamEvaluation?.executed&&step.pamEvaluation.apiPhase==='close');
 expect(result).toBeGreaterThan(trigger);expect(rows.length).toBeGreaterThan(0);for(const step of rows){expect(step.from).toBe(owner);expect(step.to).toBe(owner);expect(step.pamEvaluation.moduleHook).toBe('pam_sm_close_session');expect(step.pamEvaluation.rawReturn).toBe('PAM_SUCCESS');}
 expectReturn(model.steps()[result].pamEvaluation,'close',0,'PAM_SUCCESS');expect(end(model).workstation.pamResults.close).toBe('PAM_SUCCESS');expect(end(model).workstation.targetProcessExited).toBe(true);expect(end(model).workstation.session).toBe(true);expect(end(model).workstation.unrelatedSession).toBe('other-user-session-retained');

 const nss=SSHLAB_SCENARIOS.find(model=>model.id==='lab-ssh-sssd-success').steps().find(step=>step.sshStage==='nss-map');expect(nss.pamEvaluation.moduleHook).toBeNull();
