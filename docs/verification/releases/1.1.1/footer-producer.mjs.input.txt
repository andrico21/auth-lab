const root="/home/andrico/auth-lab-held-objects",out="/home/andrico/auth-lab-held-objects/docs/verification/releases/1.1.1/";
let browser;
const report={kind:'release1.1.1-native-visible-footer',source:"d69323ff1f26424a9185cd34cb34b0643e68f993846e6c2377fa1408ae96fbcb",standalone:{"path":"dist/auth-flow-studio.html","sha256":"7255c2da7ae2cf7aa0fa0116b291e1605ce3706075fa1d753780f91f90b78e13","bytes":5583633},startedAt:new Date().toISOString(),errors:[]};
